        const dynamicFields = document.getElementById('dynamicFields');


        attendeeSelect.addEventListener('change', function() {

            const val = this.value;

            if (val !== "" && val !== "0") {

                dynamicFields.classList.remove('hidden');

            } else {

                dynamicFields.classList.add('hidden');

            }

        });


        // 5. Robust Dual Form Submission & Feedback Logic

        function handleFormSubmit(event) {

            window.submitted = true;

            

            // 採用原生表單多路徑提交，並兼顧跨域與即時切換體驗

            const form = document.getElementById('weddingForm');

            const formData = new FormData(form);

            

            // 透過 fetch no-cors 強制送出表單至 Google 伺服器

            fetch(form.action, {

                method: 'POST',

                body: formData,

                mode: 'no-cors'

            }).then(() => {

                showThankYouMessage();

            }).catch(err => {

                showThankYouMessage();

            });


            // 確保畫面上立即切換為感謝訊息

            setTimeout(() => {

                showThankYouMessage();

            }, 500);

        }


        function showThankYouMessage() {

            const form = document.getElementById('weddingForm');

            const thankYouCard = document.getElementById('thankYouCard');

            if (form && thankYouCard) {

                form.classList.add('hidden');

                thankYouCard.classList.remove('hidden');

            }

        }
